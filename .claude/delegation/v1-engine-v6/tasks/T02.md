## Interface (implement exactly; `window.ValoAim` / `module.exports`)
```js
ValoAim.VERSION = "1.0.0"
ValoAim.HEAD_DEG = 0.716
ValoAim.FOV_H = 103
ValoAim.degPerCm(cm360) -> 360 / cm360
ValoAim.screenX(yawDeg, widthPx) -> px or null if |yaw| >= 90   // CONTRACTS projection formula
ValoAim.screenY(pitchDeg, widthPx, heightPx) -> px              // same focal length as x (square pixels)

ValoAim.levelsAround(cmCentre, n = 5, spanOct = 0.6) -> number[]   // log-spaced, ascending, centre included when n odd, each clamped to [8, 160]
ValoAim.schedule({ levels, perLevel: { micro, flick, turn, track }, warmupPerBlock = 2, seed, session = "s1", blockOrder? })
   -> blocks: [{ block, cm, specs: [spec...] }]
   // one block per level; block order = rows of a balanced Latin square for levels.length chosen by seed
   // (Williams design when n is even; for odd n use the seed's row and its mirror alternately);
   // inside a block tasks are interleaved in seeded random order; the first warmupPerBlock specs of each block have warmup:true;
   // spec.order counts across the whole session; spec.id = session + "-" + order. Geometry per CONTRACTS "Task geometry".
ValoAim.nextLevel(candidates, stats, density) -> cm
   // adaptive choice for an extra block: stats = { [cm]: nTrials }, density(cm) -> current posterior density of the optimum (number >= 0).
   // Score = density(cm) * (1 / sqrt(1 + n)) ; return the argmax (ties -> smaller |log(cm/median of candidates)|).

ValoAim.features(traj, spec) -> {
   time, hit, misses, overshoots, endErrDeg, nSub, peakCmS, pathEff, lifts, liftTimes: number[], primaryAmpCm, primaryErrCm,
   onTarget /* track only: fraction of samples with |crosshair - target| < targetDeg/2 */ }
   // Rules: crosshair yaw(t) = x(t) * 360 / traj.cm ; pitch similarly from y.
   // velocity by central differences on samples, then a 15 ms moving-average; lifts = gaps >= 0.06 s in t;
   // submovements = count of velocity peaks above 2 cm/s separated by a minimum below 30% of the smaller peak (count across lifts);
   // primary submovement = first one; primaryErrCm = signed position at its end (first velocity minimum after first peak) minus target position in cm;
   // overshoots = number of times crosshair yaw crosses past the target centre moving away from the start side by more than targetDeg/2;
   // misses = clicks with hit:false before the hit; pathEff = |net displacement| / total path length (1 = straight);
   // time: static = traj.done if hit else 4 with hit:false; track = per CONTRACTS.
ValoAim.toTrial(spec, feat, extra?) -> trial record (CONTRACTS)   // warmup trials keep warmup:true
ValoAim.dpiCheck(samples, statedDpi) -> { countsPerCm, effDpi, ratio, cv, consistent, accelSuspect }
   // samples: [{ counts, cm, durS }] from "move the mouse along a credit card (8.56 cm)" repeats.
   // countsPerCm = median(counts/cm); ratio = effDpi/statedDpi; cv = sd/mean of counts/cm; consistent = cv < 0.05;
   // accelSuspect = |corr(counts/cm, cm/durS)| > 0.6 with >= 3 samples (speed-dependent gain = acceleration).
ValoAim.padCheck(swipes) -> { value, sd, n }   // swipes: [{ cm }] edge-to-edge; value = median, sd = sd/sqrt(n)
ValoAim.reachCheck(swipes) -> { value, sd, n } // comfortable one-direction reach; value = median
ValoAim.estimatePersonal(records, extras) -> personal (CONTRACTS)
   // records: [{ spec, feat }] (non-warmup only). extras: { dpi: dpiCheck result, pad: padCheck, reach: reachCheck }.
   // sigma0: from micro trials, fit primaryErrCm variance = sigma0² + k²·A² by least squares on (A², err²) (bin by A, >= 8 trials); sigma0 = sqrt(max(intercept, 1e-6)); sd via bootstrap (200, seeded).
   // tSub: median duration of corrective submovements (2nd+) on micro/flick trials.
   // vPeak: 90th percentile of peakCmS over turn trials.
   // tLift: median of liftTimes over all trials.
   // reachCm, padCm, countsPerCm: from extras when present.
   // accel: from dpiCheck.
```

## Decisions already made
- No DOM access anywhere in aim.js (the UI task does rendering). It must load in Node.
- Seeded RNG only (copy the `rng` algorithm from `model.js`).
- Do not import model.js or sim/player.js from aim.js.
- Tests must build their own small synthetic trajectories inline (e.g. a straight minimum-jerk path with one correction and one lift gap). Do not depend on sim/player.js (being written in parallel).
